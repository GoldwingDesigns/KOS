@lazyglobal off.

{
	global UTL is lexicon(
		"slave init", 		    slaveInit@,
		"slave reboot", 		slaveReboot@,
		"has file",		    	hasFile@,
        "has file contains",	hasFileContains@,
        "list file contains",   listFileContains@,
		"delete file",		    deleteFile@,
		"retrieve file",    	retrieveFile@,
		"update libfiles",	    updateLibFiles@,
		"update bootfiles",     updateBootFile@,
		"reboot",			    reboot@
	).
	function slaveInit {
		parameter cpu,
				rebootfile,
				slavevol.
		SET VOLUME(slavevol):NAME TO cpu:TAG+"_Disk".
		copypath("Master_Disk:/",cpu:TAG+"_Disk:/").
		slaveReboot(cpu,rebootfile,slavevol).
		return true.
	}

	function slaveReboot {
		parameter cpu,
				rebootfile,
				rebootfilevol.
		cpu:DEACTIVATE().
		if rebootfile:length > 0 {
			if hasfile(rebootfile,rebootfilevol) {
				copypath(rebootfilevol + ":/" + rebootfile,"").
				set cpu:bootfilename to rebootfile. }
		}
		wait 2.
		cpu:ACTIVATE().
		return true.
	}
		
	function hasFile {
		//taken from u/DeOriginale69
		Parameter file_name,
				volume_number.
		local filematch to FALSE.
		local tempvol to CORE:VOLUME:NAME.
		Switch to volume_number.
		List FILES In allFiles.
		For file In allFiles {
		If file:Name = file_name {
				Set filematch to True.
                Switch to tempvol.
                return filematch.
				} 
			}
		Switch to tempvol.
		Return filematch.
	} 

    function hasFileContains {
		//taken from u/DeOriginale69
		Parameter file_name,
				volume_number.
		local filematch to FALSE.
		local tempvol to CORE:VOLUME:NAME.
		Switch to volume_number.
		List FILES In allFiles.
		For file In allFiles {
		If file:Name:contains(file_name) {
				Set filematch to True.
                Switch to tempvol.
                return filematch.
				} 
			}
		Switch to tempvol.
		Return filematch.
	} 
    function hasFileContains {
		//taken from u/DeOriginale69
		Parameter file_name,
				volume_number.
		local filelist to list().
		local tempvol to CORE:VOLUME:NAME.
		Switch to volume_number.
		List FILES In allFiles.
		For file In allFiles {
		If file:Name:contains(file_name) {
				filelist:add(file:Name).
				} 
			}
		Switch to tempvol.
		Return filelist.
	} 

	function deleteFile {
		parameter filename,
				volnum.
		if hasFile(filename,volnum) deletepath(volnum+":/"+filename). 
		return true.
	}

	function retrieveFile {
		parameter filename,
				volnum.
		if hasFile(filename,volnum) copypath(volnum+":/"+filename,"1:").
		return true.
	}

	function updateLibFiles {
		IF HOMECONNECTION:ISCONNECTED{
			FOR f in LIBFILES {
				copypath("0:/"+f,"1:").
			}
				gui_msg_add("Library Files Updated.").
		} ELSE {
			gui_msg_add("Ship not connected to Kerbin.").
		}
		return true.
	}

	function reloadLibFiles {
		gui_msg_add("Lib Files Reloaded.").
		return true.
	}

	function updateBootFile {
		IF HOMECONNECTION:ISCONNECTED{
			FOR f in LIBFILES {
				copypath("0:/"+f,"1:").
			}
			gui_msg_add("Boot Files Updated.").
		} ELSE {
		gui_msg_add("Ship not connected to Kerbin.").
		}
		return true.
	}

	function reboot {
		reboot.
	}
}
